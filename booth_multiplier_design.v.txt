`timescale 1ns / 1ps
module low_power_booth_multiplier(
    input clk,
    input signed [7:0] A,
    input signed [7:0] B,
    output reg signed [15:0] P
);

integer i;
reg signed [8:0] M;
reg signed [16:0] result;
reg signed [8:0] temp;
reg [1:0] booth_bits;
reg prev_bit;

always @(posedge clk) begin
    M = {A[7], A};   // sign extend
    result = 0;
    prev_bit = 0;

    for (i = 0; i < 4; i = i + 1) begin
        booth_bits = {B[2*i+1], B[2*i]};

        case ({booth_bits, prev_bit})
            3'b000, 3'b111: temp = 0;
            3'b001, 3'b010: temp = M;
            3'b011:         temp = M <<< 1;
            3'b100:         temp = -(M <<< 1);
            3'b101, 3'b110: temp = -M;
        endcase

        result = result + (temp <<< (2*i));
        prev_bit = B[2*i+1];
    end

    P <= result[15:0];
end

endmodule
